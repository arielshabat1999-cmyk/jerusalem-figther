/* V151 — live Player Rank objective tracker. Five-objective ranks are fully functional. */
;(()=>{
if(!window.PlayerRank)return;
let run={kills:0,upgrades:0,survival:0,noDamage:0,lastDistance:0,lastScore:0,lastLevel:1};
const originalReset144=reset,originalKill144=kill,originalHurt144=hurt,originalUpdate144=update;
function objectives144(){const d=window.PlayerRank.get(),c=window.PLAYER_RANK_CONFIG,r=c&&c.ranks&&c.ranks[d.rank];return r?r.objectives.filter(o=>o.enabled!==false):[]}

window.PlayerRankTrack={upgrade(){run.upgrades++;setType144('upgrades',run.upgrades,'single_run')},eliteKill(){addType144('elite_kills',1,'career')},bossKill(){addType144('boss_kills',1,'career')},snapshot:()=>({...run})};
