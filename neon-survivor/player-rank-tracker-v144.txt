/* V160 — live objective authority. Gameplay level comes from the XP system HUD because XP level L is private to xp-builds-v87 closure. */
;(()=>{
let attached=false,run=null,base=null;
const fresh=()=>({kills:0,upgrades:0,survival:0,noDamage:0,lastDistance:0,lastLevel:1});
function apiReady(){return !!(window.PlayerRank&&window.PLAYER_RANK_CONFIG)}
function objectives(){if(!apiReady())return[];const d=PlayerRank.get(),r=PLAYER_RANK_CONFIG.ranks?.[d.rank];return r?.objectives?.filter(o=>o.enabled!==false)||[]}
function set(type,value,scope){for(const o of objectives())if(o.type===type&&(!scope||o.scope===scope))PlayerRank.setProgress(o.id,value)}
function add(type,value=1,scope){for(const o of objectives())if(o.type===type&&(!scope||o.scope===scope))PlayerRank.addProgress(o.id,value)}
function resetRun(){run=fresh()}
function gameplayLevel(){const el=document.getElementById('xp87l');if(el){const m=(el.textContent||'').match(/LEVEL\s*(\d+)/i);if(m)return Math.max(1,+m[1]||1)}const modal=document.getElementById('levelSub87');if(modal){const m=(modal.textContent||'').match(/LEVEL\s*(\d+)/i);if(m)return Math.max(1,+m[1]||1)}return run?.lastLevel||1}
function attach(){if(attached||!apiReady())return false;if(typeof reset!=='function'||typeof kill!=='function'||typeof hurt!=='function'||typeof update!=='function')return false;attached=true;resetRun();base={reset,kill,hurt,update};
reset=function(){const r=base.reset.apply(this,arguments);resetRun();return r};
kill=function(e){const r=base.kill.apply(this,arguments);run.kills++;add('kills',1,'career');set('kills',run.kills,'single_run');if(e&&(e.type==='elite'||e.elite===true||e.isElite===true))add('elite_kills',1,'career');if(e&&(e.type==='boss'||e.boss===true||e.isBoss===true))add('boss_kills',1,'career');return r};
hurt=function(n){const hp=p?.hp,sh=p?.shield,r=base.hurt.apply(this,arguments);if(p&&((hp!=null&&p.hp<hp)||(sh!=null&&p.shield<sh)))run.noDamage=0;return r};
update=function(dt){const wasPlayable=state==='play'||state==='sector'||state==='boss',r=base.update.apply(this,arguments);if(!wasPlayable||!p)return r;const sec=Math.max(0,+dt||0);run.survival+=sec;run.noDamage+=sec;set('survival_seconds',run.survival,'single_run');set('no_damage_seconds',run.noDamage,'single_run');const dist=Math.max(0,+distance||0);set('distance',dist,'single_run');if(dist>run.lastDistance){add('distance',dist-run.lastDistance,'career');run.lastDistance=dist}const sc=Math.max(0,+score||0);set('score',sc,'single_run');const lv=gameplayLevel();set('run_level',lv,'single_run');if(lv>run.lastLevel){run.upgrades+=lv-run.lastLevel;run.lastLevel=lv;set('upgrades',run.upgrades,'single_run')}return r};
window.PlayerRankTrack={upgrade(){run.upgrades++;set('upgrades',run.upgrades,'single_run')},eliteKill(){add('elite_kills',1,'career')},bossKill(){add('boss_kills',1,'career')},gameplayLevel,snapshot:()=>({...run,level:gameplayLevel()}),attached:()=>attached};console.log('OBJECTIVE AUTHORITY V160 ATTACHED');return true}
if(!attach()){window.addEventListener('player-rank-ready',attach,{once:true});let tries=0;const timer=setInterval(()=>{if(attach()||++tries>100)clearInterval(timer)},50)}
})();
