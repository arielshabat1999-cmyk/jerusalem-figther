/* V121 — Falcon modular ship renderer: preload once, then render synchronously in Garage + gameplay. */
;(()=>{try{
const VERSION='20260918v129',IDS=['falcon'];
const META={
  falcon:{
    anchor:{x:.5,y:.5},
    enginePoints:[{x:.347,y:.850},{x:.653,y:.850}],
    muzzlePoints:[{x:.3125,y:.210},{x:.6875,y:.210}],
    trailPoint:{x:.5,y:.889}
  }
}
const LAYERS=['hull-base','color-mask','livery-mask','lights-emissive','cockpit','engine-flame','shadow-ao'];
const cache=new Map(),prepared=new Map();
const ENGINE_FRAME_INDEX={blue:0,gold:1,red:2,purple:3,green:4};
const ENGINE_SPRITE_PATH='./assets/ships/modular/falcon/engine-options.webp?v='+VERSION;
const ENGINE_OPTIONS={
  blue:{id:'blue',name:'ION BLUE',color:'#35d8ff'},
  gold:{id:'gold',name:'SOLAR GOLD',color:'#ffb52d'},
  red:{id:'red',name:'CRIMSON',color:'#ff3a2f'},
  purple:{id:'purple',name:'NEON VIOLET',color:'#b34dff'},
  green:{id:'green',name:'PLASMA GREEN',color:'#35ff9b'}
};
const SHIP_COLOR_OPTIONS={
  silver:{id:'silver',name:'SILVER',color:''},
  blue:{id:'blue',name:'COBALT',color:'#2f7fff'},
  red:{id:'red',name:'CRIMSON',color:'#e53935'},
  gold:{id:'gold',name:'GOLD',color:'#d9a441'},
  purple:{id:'purple',name:'VIOLET',color:'#8b4dff'}
};
function engineOption(id){return ENGINE_OPTIONS[id]||ENGINE_OPTIONS.blue}
function shipColorOption(id){return SHIP_COLOR_OPTIONS[id]||SHIP_COLOR_OPTIONS.silver}

function pathFor(id,layer){return './assets/ships/modular/'+id+'/'+layer+'.webp?v='+VERSION}
function loadImage(src,optional=false){
  return new Promise((resolve,reject)=>{
    const im=new Image();
    im.decoding='async';
    im.onload=async()=>{
      try{
        if(im.decode)await im.decode();
        resolve(im);
      }catch(e){
        console.error('MODULAR ASSET DECODE FAILED',src,e);
        optional?resolve(null):reject(new Error('MODULAR ASSET DECODE FAILED '+src));
      }
    };
    im.onerror=()=>{
      console.error('MODULAR ASSET LOAD FAILED',src);
      optional?resolve(null):reject(new Error('MODULAR ASSET LOAD FAILED '+src));
    };
    im.src=src;
  });
}
async function load(id='falcon'){
  if(!IDS.includes(id))id='falcon';
  if(cache.has(id))return cache.get(id);
  const job=(async()=>{
    const [rows,engineSprite]=await Promise.all([
      Promise.all(LAYERS.map(async layer=>[layer,await loadImage(pathFor(id,layer),layer!=='hull-base')])),
      loadImage(ENGINE_SPRITE_PATH,true)
    ]);
    const layers=Object.fromEntries(rows);
    if(!layers['hull-base'])throw new Error('Falcon hull-base missing');
    return{id,meta:META[id],layers,engineSprite};
  })();
  cache.set(id,job);
  return job;
}
function tintMask(img,color){
  if(!img||!color)return null;
  const cv=document.createElement('canvas'),w=img.naturalWidth||img.width,h=img.naturalHeight||img.height;
  cv.width=w;cv.height=h;
  const c=cv.getContext('2d');
  c.drawImage(img,0,0,w,h);
  c.globalCompositeOperation='source-in';
  c.fillStyle=color;c.fillRect(0,0,w,h);
  c.globalCompositeOperation='source-over';
  return cv;
}
function compositeKey(id,opt={}){return id+'|'+(opt.color||'')+'|'+(opt.livery||'')}
function prepare(id='falcon',opt={}){
  if(!IDS.includes(id))id='falcon';
  const key=compositeKey(id,opt);
  if(prepared.has(key))return prepared.get(key);
  const job=(async()=>{
    const a=await load(id),base=a.layers['hull-base'];
    const w=base.naturalWidth||base.width,h=base.naturalHeight||base.height;
    const cv=document.createElement('canvas');cv.width=w;cv.height=h;
    const c=cv.getContext('2d');
    if(a.layers['shadow-ao'])c.drawImage(a.layers['shadow-ao'],0,0,w,h);
    c.drawImage(base,0,0,w,h);
    if(opt.color){
      /* Guaranteed hull recolor using the real hull alpha.
         This no longer depends on the old mask asset being usable. */
      c.save();
      c.globalCompositeOperation='source-atop';
      c.globalAlpha=.58;
      c.fillStyle=opt.color;
      c.fillRect(0,0,w,h);
      c.restore();
      /* Put a little original shading/detail back over the tint. */
      c.save();
      c.globalAlpha=.34;
      c.drawImage(base,0,0,w,h);
      c.restore();
    }
    if(opt.livery&&a.layers['livery-mask']){
      const t=tintMask(a.layers['livery-mask'],opt.livery);
      c.save();c.globalCompositeOperation='source-atop';c.globalAlpha=.88;c.drawImage(t,0,0,w,h);c.restore();
    }
    if(a.layers.cockpit)c.drawImage(a.layers.cockpit,0,0,w,h);
    return{id,meta:a.meta,layers:a.layers,body:cv,w,h};
  })();
  prepared.set(key,job);
  return job;
}
function getPrepared(id='falcon',opt={}){
  const p=prepared.get(compositeKey(IDS.includes(id)?id:'falcon',opt));
  if(!p||typeof p.then!=='function')return null;
  return p.__value||null;
}
async function prepareTracked(id='falcon',opt={}){
  const key=compositeKey(id,opt);
  const p=prepare(id,opt);
  if(!p.__tracked){
    p.__tracked=true;
    p.then(v=>{p.__value=v;return v});
  }
  return p;
}
function drawTintedLayer(ctx,img,color,x,y,w,h,alpha=1,glow=0){
  if(!img)return;
  const tint=tintMask(img,color);
  if(!tint)return;
  ctx.save();ctx.globalCompositeOperation='lighter';ctx.globalAlpha=alpha;
  if(glow){ctx.shadowBlur=glow;ctx.shadowColor=color}
  ctx.drawImage(tint,x,y,w,h);ctx.restore();
}
function drawEngineVariant(ctx,a,x,y,w,h,engineId,time=0){
  const sprite=a.engineSprite;
  if(!sprite)return;
  const frame=ENGINE_FRAME_INDEX[engineId]??0;
  const sw=(sprite.naturalWidth||sprite.width)/5,sh=(sprite.naturalHeight||sprite.height);
  const pulse=.94+.06*Math.sin(time*11);
  const color=engineOption(engineId).color;

  /* The sprite contains ONE real generated flame per frame.
     Draw it twice, centered on the Falcon's real engine points. */
  const fw=w*.19,fh=h*.30;

  ctx.save();
  ctx.globalCompositeOperation='lighter';
  ctx.globalAlpha=pulse;
  ctx.shadowBlur=Math.max(12,w*.05);
  ctx.shadowColor=color;

  for(const p of a.meta.enginePoints){
    const px=x+p.x*w;
    const py=y+p.y*h;
    ctx.drawImage(sprite,frame*sw,0,sw,sh,px-fw*.5,py-fh*.03,fw,fh);
  }
  ctx.restore();
}
function drawSync(ctx,opt={}){
  const id=IDS.includes(opt.shipId)?opt.shipId:'falcon';
  const a=getPrepared(id,opt);
  if(!a)return false;
  const scale=opt.scale||1,h=(opt.height||100)*scale,w=h*(a.w/a.h);
  const cx=opt.x||0,cy=opt.y||0,x=-w*.5,y=-h*.5,time=opt.time||0;
  const engine=engineOption(opt.engineId||'blue');
  ctx.save();ctx.translate(cx,cy);ctx.rotate(opt.rotation||0);ctx.globalAlpha=opt.alpha??1;

  /* Engine choice affects ONLY the engine fire. Draw it behind the hull. */
  if(opt.showEngineFlame!==false)drawEngineVariant(ctx,a,x,y,w,h,engine.id,time);

  ctx.drawImage(a.body,x,y,w,h);

  /* Ship emissive lighting stays the Falcon's original cyan/white lighting. */
  if(opt.showGlow!==false&&a.layers['lights-emissive']){
    const pulse=.76+.16*Math.sin(time*7);
    ctx.save();ctx.globalCompositeOperation='lighter';ctx.globalAlpha=pulse;
    ctx.shadowBlur=11;ctx.shadowColor='#55edff';
    ctx.drawImage(a.layers['lights-emissive'],x,y,w,h);
    ctx.restore();
  }
  ctx.restore();
  return{w,h,meta:a.meta,modular:true};
}
async function renderCanvas(canvas,opt={}){
  await prepareTracked(opt.shipId||'falcon',opt);
  const r=canvas.getBoundingClientRect(),ratio=Math.min(devicePixelRatio||1,2);
  const w=Math.max(1,Math.round(r.width)),h=Math.max(1,Math.round(r.height));
  canvas.width=Math.round(w*ratio);canvas.height=Math.round(h*ratio);
  const c=canvas.getContext('2d');c.setTransform(ratio,0,0,ratio,0,0);c.clearRect(0,0,w,h);
  const targetH=opt.height||h*.78;
  return drawSync(c,{...opt,x:opt.x??w/2,y:opt.y??h*.47,height:targetH});
}
function clearCompositeCache(){prepared.clear()}
const API={version:129,ids:IDS,meta:META,engineOptions:ENGINE_OPTIONS,engineOption,shipColorOptions:SHIP_COLOR_OPTIONS,shipColorOption,load,prepare:prepareTracked,drawSync,renderCanvas,clearCompositeCache};
window.__SHIP_MODULAR_V120=API;
let startupColor='';
try{startupColor=shipColorOption(localStorage.NSFalconColor||'silver').color}catch(_){}
Promise.all([
  prepareTracked('falcon'),
  prepareTracked('falcon',{color:startupColor})
]).then(()=>{
  try{parent.postMessage({source:'ship-modular-v120',status:'MODULAR SHIP READY'},'*')}catch(_){}
}).catch(e=>{
  console.error('Falcon modular preload failed',e);
  try{parent.postMessage({source:'ship-modular-v120',status:'MODULAR SHIP ERROR'},'*')}catch(_){}
});
}catch(e){
 console.error('SHIP MODULAR V121 SAFE FAIL',e);
 try{parent.postMessage({source:'ship-modular-v120',status:'MODULAR SHIP ERROR'},'*')}catch(_){}
}})();
