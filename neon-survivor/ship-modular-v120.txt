/* V130 — Falcon renderer using pre-rendered hull + engine assets only. No runtime tint masks. */
;(()=>{try{
const VERSION='20260918v131',IDS=['falcon'];
const META={
  falcon:{
    anchor:{x:.5,y:.5},
    enginePoints:[{x:.347,y:.850},{x:.653,y:.850}],
    muzzlePoints:[{x:.3125,y:.210},{x:.6875,y:.210}],
    trailPoint:{x:.5,y:.889}
  }
};
const ENGINE_OPTIONS={
  blue:{id:'blue',name:'ION BLUE',frame:0},
  gold:{id:'gold',name:'SOLAR GOLD',frame:1},
  red:{id:'red',name:'CRIMSON',frame:2},
  purple:{id:'purple',name:'NEON VIOLET',frame:3},
  green:{id:'green',name:'PLASMA GREEN',frame:4}
};
const SHIP_COLOR_OPTIONS={
  silver:{id:'silver',name:'SILVER',frame:0},
  blue:{id:'blue',name:'COBALT',frame:1},
  red:{id:'red',name:'CRIMSON',frame:2},
  gold:{id:'gold',name:'GOLD',frame:3},
  purple:{id:'purple',name:'VIOLET',frame:4}
};
const prepared=new Map();

function engineOption(id){return ENGINE_OPTIONS[id]||ENGINE_OPTIONS.blue}
function shipColorOption(id){return SHIP_COLOR_OPTIONS[id]||SHIP_COLOR_OPTIONS.silver}
function loadImage(src){
  return new Promise((resolve,reject)=>{
    const im=new Image();im.decoding='async';
    im.onload=async()=>{try{if(im.decode)await im.decode();resolve(im)}catch(e){reject(e)}};
    im.onerror=()=>reject(new Error('ASSET LOAD FAILED '+src));
    im.src=src;
  });
}
function asset(name){return './assets/ships/modular/falcon/'+name+'?v='+VERSION}
function prepare(id='falcon'){
  if(!IDS.includes(id))id='falcon';
  if(prepared.has(id))return prepared.get(id);
  const p=(async()=>{
    const [hullSheet,engineSheet]=await Promise.all([
      loadImage(asset('hull-variants.webp')),
      loadImage(asset('engine-variants.webp'))
    ]);
    return{id,meta:META[id],hullSheet,engineSheet};
  })();
  p.then(v=>{p.__value=v;return v});
  prepared.set(id,p);
  return p;
}
function getPrepared(id='falcon'){
  const p=prepared.get(IDS.includes(id)?id:'falcon');
  return p&&p.__value?p.__value:null;
}
function drawFrame(ctx,img,frame,total,x,y,w,h,alpha=1,glowColor=''){
  if(!img)return;
  const sw=(img.naturalWidth||img.width)/total,sh=(img.naturalHeight||img.height);
  ctx.save();
  ctx.globalAlpha=alpha;
  if(glowColor){
    ctx.globalCompositeOperation='lighter';
    ctx.shadowBlur=Math.max(8,w*.045);
    ctx.shadowColor=glowColor;
  }
  ctx.drawImage(img,frame*sw,0,sw,sh,x,y,w,h);
  ctx.restore();
}
function drawSync(ctx,opt={}){
  const id=IDS.includes(opt.shipId)?opt.shipId:'falcon';
  const a=getPrepared(id);
  if(!a)return false;

  const scale=opt.scale||1,h=(opt.height||100)*scale,w=h;
  const cx=opt.x||0,cy=opt.y||0,x=-w*.5,y=-h*.5,time=opt.time||0;
  const color=shipColorOption(opt.colorId||'silver');
  const engine=engineOption(opt.engineId||'blue');
  const pulse=.92+.08*Math.sin(time*12);

  ctx.save();
  ctx.translate(cx,cy);
  ctx.rotate(opt.rotation||0);
  ctx.globalAlpha=opt.alpha??1;

  if(opt.showEngineFlame!==false){
    const glow=engine.id==='gold'?'#ffb52d':engine.id==='red'?'#ff3a2f':engine.id==='purple'?'#b34dff':engine.id==='green'?'#35ff9b':'#35d8ff';

    /* The engine sheet is an effects asset, not a full-ship-aligned frame.
       Place it lower and narrower so the flames emerge from the Falcon's rear
       instead of sitting over the middle of the hull. */
    const ew=w*.74,eh=h*.72;
    const ex=-ew*.5;
    const ey=-h*.08;
    drawFrame(ctx,a.engineSheet,engine.frame,5,ex,ey,ew,eh,pulse,glow);
  }

  /* Hull is always drawn last so it cleanly masks the root of the flames. */
  drawFrame(ctx,a.hullSheet,color.frame,5,x,y,w,h,1,'');

  ctx.restore();
  return{w,h,meta:a.meta,modular:true,colorId:color.id,engineId:engine.id};
}
async function renderCanvas(canvas,opt={}){
  await prepare(opt.shipId||'falcon');
  const r=canvas.getBoundingClientRect(),ratio=Math.min(devicePixelRatio||1,2);
  const w=Math.max(1,Math.round(r.width)),h=Math.max(1,Math.round(r.height));
  canvas.width=Math.round(w*ratio);canvas.height=Math.round(h*ratio);
  const c=canvas.getContext('2d');
  c.setTransform(ratio,0,0,ratio,0,0);
  c.clearRect(0,0,w,h);
  const targetH=opt.height||h*.78;
  return drawSync(c,{...opt,x:opt.x??w/2,y:opt.y??h*.47,height:targetH});
}
function clearCompositeCache(){prepared.clear()}
const API={version:131,ids:IDS,meta:META,engineOptions:ENGINE_OPTIONS,engineOption,shipColorOptions:SHIP_COLOR_OPTIONS,shipColorOption,prepare,drawSync,renderCanvas,clearCompositeCache};
window.__SHIP_MODULAR_V120=API;

prepare('falcon').then(()=>{
  try{parent.postMessage({source:'ship-modular-v120',status:'MODULAR SHIP READY'},'*')}catch(_){}
}).catch(e=>{
  console.error('Falcon V131 asset preload failed',e);
  try{parent.postMessage({source:'ship-modular-v120',status:'MODULAR SHIP ERROR'},'*')}catch(_){}
});
}catch(e){
  console.error('SHIP MODULAR V131 SAFE FAIL',e);
  try{parent.postMessage({source:'ship-modular-v120',status:'MODULAR SHIP ERROR'},'*')}catch(_){}
}})();
