/* V57 — deterministic 1..20 enemy wave cycle with escalating threat tiers */
;(()=>{
let wave57=1,tier57=0,cooldown57=.45,waveSerial57=0;
const spawnOne57=enemy;

function regular57(){return E.filter(e=>e&&!e.boss&&!e.boss5)}
function strengthen57(e){
  if(!e||e.boss||e.boss5)return;
  const hpMul=1+tier57*.42,dmgMul=1+tier57*.20,fireMul=Math.max(.48,1-tier57*.075);
  e.hp*=hpMul;e.max*=hpMul;e.dmg*=dmgMul;e.shotDmg*=dmgMul;e.fireRate*=fireMul;
  e.val=Math.round((e.val||10)*(1+tier57*.18));e.credit=Math.round((e.credit||4)*(1+tier57*.10));
  e.threatTier57=tier57;
  if(e.flow10){e.flow10.hold*=Math.min(1.55,1+tier57*.07);e.flow10.fire*=fireMul}
}
function formation57(e,i,count){
  if(!e||!e.flow10)return;
  const cols=Math.min(5,count),rows=Math.ceil(count/cols),row=Math.floor(i/cols),col=i%cols;
  const used=row===rows-1?count-row*cols:cols;
  const x=W*(used===1?.5:.12+.76*(col/Math.max(1,used-1)));
  const y=-70-row*48-R(0,12),f=e.flow10;
  e.x=x;e.y=y;f.x=x;f.y=y;f.anchorX=x;
  f.targetY=H*(.17+(row%4)*.095+R(0,.025));
  f.seed=waveSerial57*.71+i*1.37+R(0,.3);
}
function spawnWave57(){
  const count=wave57;waveSerial57++;
  for(let i=0;i<count;i++){
    const before=E.length;spawnOne57();
    const e=E[E.length-1];if(E.length===before||!e)continue;
    strengthen57(e);formation57(e,i,count);
  }
  flashStage('WAVE '+count+(tier57?' · THREAT '+(tier57+1):''));
  popText(W*.5,H*.23,count+' HOSTILE'+(count===1?'':'S'),'#ff7a78',17);
  if(wave57>=20){wave57=1;tier57++}else wave57++;
  cooldown57=.8;
}

const update57base=update;
update=function(dt){
  spawn=999;
  update57base(dt);
  spawn=999;
  if(!p||state!=='play')return;
  /* V53 reserves the last pixels of a sector for clearing before the boss. */
  if(distance>=stage*STAGE_M-3)return;
  cooldown57-=dt;
  if(cooldown57<=0&&regular57().length===0)spawnWave57();
};

const checkpoint57base=checkpoint;
checkpoint=function(){checkpoint57base();if(state==='play'){spawn=999;cooldown57=.65}};

const reset57base=reset;
reset=function(){wave57=1;tier57=0;cooldown57=.45;waveSerial57=0;reset57base();spawn=999};
})();
