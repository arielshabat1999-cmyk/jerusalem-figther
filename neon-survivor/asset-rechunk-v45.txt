/* V45 base64 rechunk fix: re-slice atlas chunks on 4-char boundaries so per-chunk atob is valid */
;(()=>{
try{
  const src=[window.B17C0,window.B17C1,window.B17C2,window.B17C3,window.B17C4,window.B17C5];
  if(src.some(v=>typeof v!=='string'||v.length<100)) throw new Error('missing source chunk');
  const all=src.join('').replace(/\s+/g,'');
  if(!all.length || all.length%4!==0) throw new Error('combined base64 length '+all.length+' is not divisible by 4');
  const step=Math.floor((all.length/6)/4)*4;
  window.B17C0=all.slice(0,step);
  window.B17C1=all.slice(step,step*2);
  window.B17C2=all.slice(step*2,step*3);
  window.B17C3=all.slice(step*3,step*4);
  window.B17C4=all.slice(step*4,step*5);
  window.B17C5=all.slice(step*5);
  const out=[B17C0,B17C1,B17C2,B17C3,B17C4,B17C5];
  if(out.some(v=>v.length%4!==0)) throw new Error('rechunk alignment failed');
  try{parent.postMessage({source:'cinematic-v42',status:'RECHUNK OK'},'*')}catch(_){}
}catch(e){
  console.error('V45 rechunk failed',e);
  try{parent.postMessage({source:'cinematic-v42',status:'RECHUNK ERROR'},'*')}catch(_){}
}
})();
